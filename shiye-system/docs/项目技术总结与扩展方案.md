## 🔄 三、多省份扩展方案

### 3.1 架构升级设计

#### 方案A：独立部署（推荐初期）

**特点**：
- 每个省份独立一套系统
- 数据库独立
- 端口独立
- 独立维护

**优势**：
- 简单快速
- 互不影响
- 风险隔离
- 易于调试

**劣势**：
- 代码重复
- 维护成本高
- 功能更新需要同步

**适用场景**：
- 初期验证阶段
- 省份数量<3个
- 快速上线需求

class AnalysisService:
    """数据分析服务（支持多省份）"""

    def estimate_salary(position, province_config):
        """
        预估岗位年薪（支持省份配置）
        
        Args:
            position: 岗位对象
            province_config: 省份配置字典
        """
        # 使用省份特定的薪资映射
        city_salary_map = province_config['city_salary']
        base_salary = city_salary_map.get(position.city, 8)
        
        # ... 其他逻辑保持不变
        return base_salary
    
    @staticmethod
    def estimate_competition_ratio(position, province_config):
        """
        预估竞争比（支持省份配置）

        # 使用省份特定的竞争比映射
        city_competition_map = province_config['city_competition']

    category VARCHAR(20)                  -- career/civil_service

CREATE TABLE provinces (
    id INTEGER PRIMARY KEY,
    code VARCHAR(20) UNIQUE,
    name VARCHAR(50),
    exam_types TEXT                       -- JSON: ["shiye", "shengkao"]
);

-- 3. 岗位表（升级版）
CREATE TABLE positions (
    id INTEGER PRIMARY KEY,
    province VARCHAR(20),                 -- 省份
    exam_type VARCHAR(20),                -- 考试类型
    year INTEGER,
    -- ... 其他字段
    
    FOREIGN KEY (province) REFERENCES provinces(code)
);

-- 4. 复合索引
CREATE INDEX idx_positions_province_exam ON positions(province, exam_type, year);
